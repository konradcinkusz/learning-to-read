% preamble-firstwords.tex
% -----------------------------------------------------------------------
% Lo propio de «First Words» (firstwords.tex, firstwords-bw.tex), el
% cuaderno de primeras palabras en inglés, cargado DESPUÉS de preamble.tex
% y de preamble-palabras.tex: la página, las cajas de actividad y el pie
% son los del cuaderno de primeras palabras en español (ver
% notes/06-first-words.md); aquí solo cambia la tarjeta de la palabra.
%
% En español, la tarjeta enseña la palabra partida en sílabas y entera.
% En inglés se lee por sonidos, así que la tarjeta enseña la palabra
% entera con un "botón" debajo de cada sonido (sound buttons, como en
% los colegios ingleses): un punto si el sonido es una sola letra (c, a,
% t), una raya si son varias (sh, ee, ck), y un arco que une la vocal con
% la e del final si es una e mágica (cake). Se lee tocando cada botón y
% diciendo su sonido, y luego la palabra de corrido.
% -----------------------------------------------------------------------

\usetikzlibrary{calc}

% Sin ligaduras: Andika junta ff, fi y fl en una sola letra, con una sola
% barra, y en este cuaderno cada letra cuenta -- off se lee o-ff, y esas
% dos efes tienen que verse como dos, igual que en la tabla de sonidos de
% cualquier colegio. La misma letra de preamble.tex, solo sin "liga" (los
% guiones y las comillas de TeX, -- y ``, siguen funcionando).
\setmainfont{Andika}[
  Path = fonts/andika/,
  Extension = .ttf,
  UprightFont = *-Regular,
  BoldFont = *-Bold,
  ItalicFont = *-Italic,
  BoldItalicFont = *-BoldItalic,
  Ligatures = NoCommon,
]
\setsansfont{Andika}[
  Path = fonts/andika/,
  Extension = .ttf,

% Cada grafema es un nodo pegado al anterior por la línea base
% (tools/gen_palabras.py, tarjeta_sonidos): la palabra se ve entera, como
% si fuera un solo texto, y los botones caen debajo de sus letras.
\tikzset{grafema/.style={anchor=base west, inner sep=0pt, outer sep=0pt}}

% Los botones, en unidades de la letra (em), para que sirvan igual con la
% palabra enorme del otoño que con las tres más pequeñas de la
% primavera. A 0,36 em por debajo de la línea base quedan por debajo de
% las letras que bajan (p, g, y).
\newcommand{\botonPunto}[1]{%
  \fill[colorLectura] ($(#1.base)+(0,-0.36em)$) circle (0.075em);}
\newcommand{\botonRaya}[1]{%
  \draw[colorLectura, line width=0.06em, line cap=round]
    ($(#1.base west)+(0.12em,-0.36em)$) -- ($(#1.base east)+(-0.12em,-0.36em)$);}
% La e mágica: un arco por debajo, de la vocal a la e, que pasa por
% debajo del botón de la consonante de en medio.
\newcommand{\botonArco}[2]{%
  \draw[colorLectura, line width=0.06em, line cap=round]
    ($(#1.base)+(0,-0.3em)$) .. controls +(0.05em,-0.42em) and +(-0.05em,-0.42em)
    .. ($(#2.base)+(0,-0.3em)$);}

% La tarjeta: #1 el trimestre (el tamaño, \fuentePalabra), #2 los nodos
% y los botones. La letra se elige antes del dibujo para que los em de
% los botones sean los de la palabra.
\newcommand{\tarjetaSonidos}[2]{%
  {\fuentePalabra{#1}\ajustado{\begin{tikzpicture}[baseline=(g1.base)]#2\end{tikzpicture}}\par}%
}

% Un nombre que se lee de un golpe (Toby, Amy): la palabra entera, sin
% botones, con el mismo aire debajo que una tarjeta con botones.
\newcommand{\tarjetaEntera}[2]{%
  {\fuentePalabra{#1}\ajustado{#2}\par\vspace{0.3em}}%
}

% Una tricky word, los viernes de primavera (tools/gen_palabras.py,
% lectura_tricky): con su casilla, como las palabras de la semana, pero
% dentro de un marco y sin botones -- se lee entera.
\newcommand{\palabraTricky}[1]{\mbox{\casillaGrande\hspace{2mm}%
  \fcolorbox{colorCrea}{white}{\fontsize{24}{28}\selectfont\strut\ #1\ }}}

% =========================================================================
% La tabla de sonidos (frontmatter/firstwords/sonidos.tex; las secciones
% las genera tools/gen_palabras.py en content/firstwords/generated-sonidos.tex)
% =========================================================================
% Cada sonido en grande, y debajo una palabra que lo tiene, con sus
% letras resaltadas (en blanco y negro, la negrita sola ya lo dice). Las
% casillas son cajas de ancho fijo dentro de un párrafo: salen nueve por
% línea -- los nueve sonidos de la segunda parte del otoño, una línea
% justa --, y el párrafo las parte donde haga falta. El \strut iguala el
% alto y el fondo de todos los sonidos, bajen o no de la línea (g, j, p,
% y): así las palabras de debajo quedan todas a la misma altura.
\newcommand{\resalta}[1]{\textcolor{colorLectura}{\bfseries #1}}
\newcommand{\seccionSonidos}[1]{%
  \par\vspace{4mm}{\large\bfseries #1\par}\vspace{1mm}%
  \noindent\raggedright\ignorespaces}
\newcommand{\sonido}[2]{%
  \makebox[18mm][c]{\begin{tabular}[t]{@{}c@{}}%
    {\fontsize{26}{30}\selectfont\strut #1}\\[-0.8mm]{\large #2}%
  \end{tabular}}\hspace{0.8mm plus 1mm}\ignorespaces}
\newcommand{\finSonidos}{\par}

% Un dibujo (#1, su ruta dentro de diagrams/, sin .tex) del tamaño que
% quepa en #2 de ancho por #3 de alto. Se escala cambiando la escala de
% las coordenadas, no la de la caja: escalar la caja adelgazaría las
% líneas en los dibujos pequeños, y así todas salen igual de gruesas --
% lo que importa para colorear sin salirse. Pero entonces el tamaño del
% dibujo no es proporcional a la escala: la caja de TikZ incluye el
% grueso de la línea, que no se escala. Así que se mide dos veces (a
% escala 1 y 2): el ancho es a·escala + c, y de ahí sale la escala justa
% para que quepa -- ni un punto de más, que sería una caja "overfull".
\newsavebox\cajaDibujo
\newlength\anchoUno \newlength\altoUno
\newlength\anchoDos \newlength\altoDos
\NewDocumentCommand{\dibujoColorear}{m m m}{%
  \begingroup
  \sbox\cajaDibujo{\input{diagrams/#1}\unskip}%
  \setlength\anchoUno{\wd\cajaDibujo}\setlength\altoUno{\dimexpr\ht\cajaDibujo+\dp\cajaDibujo\relax}%
  \sbox\cajaDibujo{\tikzset{every picture/.append style={scale=2}}\input{diagrams/#1}\unskip}%
  \setlength\anchoDos{\wd\cajaDibujo}\setlength\altoDos{\dimexpr\ht\cajaDibujo+\dp\cajaDibujo\relax}%
  \edef\escalaDibujo{\fpeval{min(
    (\the\dimexpr#2\relax - (2*\the\anchoUno - \the\anchoDos)) / (\the\anchoDos - \the\anchoUno),
    (\the\dimexpr#3\relax - (2*\the\altoUno - \the\altoDos)) / (\the\altoDos - \the\altoUno))}}%
  \tikzset{every picture/.append style={scale=\escalaDibujo}}%
  \input{diagrams/#1}\unskip
  \endgroup}

% El dibujo en la parte de abajo de una caja de actividad (\tcblower,
% con `centrado abajo`): tan grande como el sitio que deja la parte de
% arriba -- la altura del texto de la caja (\tcbtextheight, fija con
% `height fill`) menos lo que ya ocupa la de arriba (\tcb@upperbox, que
% tcolorbox ya ha compuesto cuando compone la de abajo), el hueco entre
% las dos (2 x middle = 4 mm) y un margen. #1 lo que se reserva debajo
% del dibujo (la línea de la adivinanza, el cartel del viernes). Si no
% queda sitio para un dibujo que se pueda colorear, es un error: la
% actividad es demasiado larga para ese día.
\newlength\altoHueco
\makeatletter
\newcommand{\calculaHueco}[2]{%
  \setlength\altoHueco{\dimexpr\tcbtextheight-\ht\tcb@upperbox-\dp\tcb@upperbox-6mm-#1\relax}%
  \ifdim\altoHueco<30mm
    \PackageError{firstwords}{No queda sitio para el dibujo #2: \the\altoHueco}{}%
  \fi}
\makeatother
\NewDocumentCommand{\dibujoHueco}{O{0pt} m}{%
  \calculaHueco{#1}{#2}%
  {\centering\dibujoColorear{#2}{\linewidth}{\altoHueco}\par}}

% La adivinanza con dibujos: tres tarjetas, una al lado de otra, en el
% sitio que quede; la respuesta es una de ellas. La tarjeta, blanca y
% con un marco fino, dice "elige una"; el dibujo, dentro, del tamaño
% que quepa.
\newlength\anchoOpcion
\newlength\altoOpcion
\newcommand{\opcionDibujo}[1]{%
  \begin{tcolorbox}[enhanced, width=\anchoOpcion, height=\altoOpcion, valign=center,
      halign=center, colback=white, colframe=colorGris!55, boxrule=0.6pt, arc=3mm,
      boxsep=0pt, left=2mm, right=2mm, top=2mm, bottom=2mm, nobeforeafter, box align=center]
    \dibujoColorear{#1}{\linewidth}{\tcbtextheight}%
  \end{tcolorbox}}
\NewDocumentCommand{\tresDibujos}{O{0pt} m m m}{%
  \calculaHueco{#1}{#2, #3, #4}%
  \setlength\anchoOpcion{0.31\linewidth}%
  \setlength\altoOpcion{\altoHueco}%
  \ifdim\altoOpcion>1.3\anchoOpcion \setlength\altoOpcion{1.3\anchoOpcion}\fi
  {\noindent\opcionDibujo{#2}\hfill\opcionDibujo{#3}\hfill\opcionDibujo{#4}\par}}

% "une" con dibujos: cada palabra, a la izquierda, con su dibujo, a la
% derecha; todas las columnas centradas en vertical (m), para que la
% palabra quede a media altura del dibujo. Los puntos de unir, en fila
% a cada lado (detrás de la palabra, delante del dibujo), y entre las
% dos filas todo el ancho que sobra, para las líneas.
% El alto de cada dibujo sale del sitio que hay en la caja (las cuatro
% filas y la instrucción), hasta un máximo. El dibujo va en una caja
% normal, que la columna m centra con la palabra de su fila (con un
% \vcenter dentro de la celda, la columna m deja de centrarla), y el
% punto de unir se sube a media altura del dibujo. El aire entre fila y
% fila va dentro de la celda, encima y debajo del dibujo: el \\[...] de
% una tabla no añade nada cuando la fila ya es más alta que su puntal.
\newlength\anchoUneDibujo
\setlength\anchoUneDibujo{55mm}
\newlength\altoUneDibujo
\newlength\separacionUne
\setlength\separacionUne{7mm}
