% it?": la ventana, la mesa con un bol grande encima, y el horno con una
% balda en la pared, por encima.

  \draw (-0.3,0) -- (16.3,0);
  % la ventana
  \filldraw[fill=white] (0.6,3.4) rectangle (3.6,5.9);
  \draw (2.1,3.4) -- (2.1,5.9); \draw (0.4,3.4) -- (3.8,3.4);
