% diagrams/firstwords/picnic.tex -- ¡merienda en el río! Grandma descansa
% en una manta de cuadros, debajo de un árbol, con la cesta de los
% bocadillos y un plato de melocotones; detrás, el río.
\begin{tikzpicture}[dibujofw]
  % el río, detrás
  \filldraw[fill=white] (-7.4,1.4) -- plot[domain=-7.4:3.6, samples=50, smooth] (\x,{2.6+0.12*sin(\x*150)}) -- (3.6,1.4) -- cycle;
  % el árbol
  \begin{scope}[shift={(5.4,0.6)}, scale=1.1] \arbolFW \end{scope}
  % la manta, con sus cuadros
  \filldraw[fill=white] (-6.4,-1.6) -- (5.6,-1.6) -- (4.6,1.6) -- (-5.4,1.6) -- cycle;
  \foreach \t in {0.25,0.5,0.75} {\draw ({-6.4+12*\t},-1.6) -- ({-5.4+10*\t},1.6);}
  \draw (-5.9,0) -- (5.1,0);
  % Grandma, sentada
  \begin{scope}[shift={(-3.4,-0.8)}] \sentadaFW{Grandma} \end{scope}
  % la cesta, con los bocadillos
  \filldraw[fill=white] (0.3,1.1) -- (0.8,2.1) -- (1.3,1.1) -- cycle;
  \filldraw[fill=white] (1.1,1.1) -- (1.7,2.2) -- (2.2,1.1) -- cycle;
  \draw (-0.1,1.3) .. controls (0.2,3.4) and (2.5,3.4) .. (2.8,1.3);
  \filldraw[fill=white] (-0.3,1.3) -- (0.1,-1.0) -- (2.6,-1.0) -- (3.0,1.3) -- cycle;
  \foreach \y in {-0.35,0.35} {\draw ({-0.3+0.4*(1.3-\y)/2.3},\y) -- ({3.0-0.4*(1.3-\y)/2.3},\y);}
  % los melocotones, en su plato
  \filldraw[fill=white] (4.4,-0.9) ellipse (1.3 and 0.35);
  \foreach \x/\y in {3.8/-0.35, 4.9/-0.35, 4.35/0.35} {\filldraw[fill=white] (\x,\y) circle (0.5);
    \draw (\x,{\y+0.45}) .. controls ({\x-0.2},\y) .. (\x,{\y-0.4});}
  \filldraw[fill=white] (4.45,0.8) .. controls (4.7,1.1) and (5.0,1.1) .. (5.15,0.95) .. controls (4.9,0.75) and (4.6,0.75) .. (4.45,0.8) -- cycle;
\end{tikzpicture}
