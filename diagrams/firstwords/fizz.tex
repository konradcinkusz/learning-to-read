% diagrams/firstwords/fizz.tex -- Bonfire Night: un cohete que sube
% chisporroteando y estalla en el cielo; cada pétalo del estallido,
% cerrado, para colorearlo de un color.
\begin{tikzpicture}[dibujofw]
  % el estallido
  \foreach \a in {0,30,...,330} {
    \begin{scope}[shift={(0,6.2)}, rotate=\a]
      \filldraw[fill=white] (0.55,0) .. controls (1.2,0.45) and (2.3,0.35) .. (2.7,0)
        .. controls (2.3,-0.35) and (1.2,-0.45) .. (0.55,0) -- cycle;
      \fill (3.15,0) circle (0.11);
    \end{scope}}
  \filldraw[fill=white] (0,6.2) circle (0.5);
  % el cohete, con su palo, y las chispas del fizz
  \begin{scope}[shift={(3.6,0.5)}, rotate=12]
    \draw (0,0) -- (0,2.2);
    \filldraw[fill=white] (-0.3,2.2) rectangle (0.3,3.8);
    \filldraw[fill=white] (-0.3,3.8) -- (0,4.4) -- (0.3,3.8) -- cycle;
    \foreach \x/\y in {-0.5/1.6, 0.45/1.2, -0.3/0.6, 0.55/0.3, -0.6/-0.1}
      {\draw (\x,{\y-0.18}) -- (\x,{\y+0.18}); \draw ({\x-0.18},\y) -- ({\x+0.18},\y);}
  \end{scope}
\end{tikzpicture}
