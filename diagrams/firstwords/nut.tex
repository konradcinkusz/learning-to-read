% diagrams/firstwords/nut.tex -- castañas de Indias (conkers) debajo del
% árbol: una hoja del castaño, una cáscara de pinchos abierta con su
% castaña, y dos castañas en el suelo. Para "completa": se colorean y
% se dibujan más.
\begin{tikzpicture}[dibujofw]
  % la hoja del castaño, con sus cinco dedos
  \foreach \a in {-64,-32,0,32,64}
    {\begin{scope}[shift={(-1.2,3.2)}, rotate=\a, scale=1.9] \hojaFW \end{scope}}
  % la cáscara, abierta: la mitad de atrás, la castaña, la mitad de delante
  \begin{scope}[shift={(-1.9,1.75)}, scale=1.25]
    \filldraw[fill=white] (-1.3,0) arc[start angle=180, end angle=0, radius=1.3] -- cycle;
    \foreach \a in {20,45,70,95,120,145,170}
      {\draw ({1.3*cos(\a)},{1.3*sin(\a)}) -- ({1.62*cos(\a)},{1.62*sin(\a)});}
    \filldraw[fill=white] (0,0.2) ellipse (0.95 and 0.85);
    \filldraw[fill=white] (0,0.72) ellipse (0.45 and 0.22);
    \filldraw[fill=white] (-1.3,0) arc[start angle=180, end angle=360, radius=1.3] -- cycle;
    \foreach \a in {190,215,240,265,290,315,340}
      {\draw ({1.3*cos(\a)},{1.3*sin(\a)}) -- ({1.62*cos(\a)},{1.62*sin(\a)});}
  \end{scope}
  % dos castañas en el suelo, con su mancha clara y su brillo
  \foreach \x/\r in {1.6/1.3, 4.3/1.1} {
    \filldraw[fill=white] (\x,\r) circle (\r);
    \filldraw[fill=white] (\x,{1.55*\r}) ellipse ({0.45*\r} and {0.25*\r});
    \draw ({\x-0.62*\r},{0.7*\r}) arc[start angle=200, end angle=245, radius=0.9*\r];}
  \draw (-4.3,0) -- (6,0);
\end{tikzpicture}
