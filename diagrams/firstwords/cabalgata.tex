% diagrams/firstwords/cabalgata.tex -- la cabalgata de Reyes (el viernes
% de "The Three Kings"): un rey en su camello, tirando caramelos.
\begin{tikzpicture}[dibujofw]
  % la cola y las patas de atrás
  \draw (-2.5,3.6) .. controls (-3.0,3.2) .. (-3.0,2.5);
  \filldraw[fill=white] (-3.0,2.5) ellipse (0.15 and 0.3);
  \foreach \x in {-1.9,1.2} {\filldraw[fill=white, rounded corners=1mm] (\x,0) rectangle ++(0.42,2.9);}
  % el cuerpo, la joroba y el cuello
  \filldraw[fill=white] (0,3.4) ellipse (2.7 and 1.2);
  \filldraw[fill=white] (-1.3,4.1) .. controls (-1.0,5.6) and (1.0,5.6) .. (1.3,4.1) -- cycle;
  \filldraw[fill=white] (1.9,3.1) .. controls (2.9,3.6) and (3.0,4.8) .. (3.3,5.9) -- (4.1,5.9)
    .. controls (3.9,4.6) and (3.6,3.2) .. (2.4,2.5) -- cycle;
  % la cabeza
  \filldraw[fill=white, rotate around={-10:(4.3,6.1)}] (4.3,6.1) ellipse (1.0 and 0.52);
  \filldraw[fill=white] (3.7,6.55) ellipse (0.14 and 0.3);
  \fill (4.1,6.3) circle (0.08); \fill (5.1,5.9) circle (0.06);
  % las patas de delante
  \foreach \x in {-1.4,1.7} {\filldraw[fill=white, rounded corners=1mm] (\x,0) rectangle ++(0.42,2.8);}
  % la manta con sus borlas
  \filldraw[fill=white] (-1.5,4.2) -- (1.5,4.2) -- (1.3,2.8) -- (-1.3,2.8) -- cycle;
  \foreach \x in {-1.0,-0.35,0.35,1.0} {\filldraw[fill=white] (\x,2.65) circle (0.15);}
  % el rey, sentado en la joroba, tirando caramelos
  \begin{scope}[shift={(0,5.0)}]
    \filldraw[fill=white] (-1.1,0) -- (1.1,0) -- (0.8,2.3) -- (-0.8,2.3) -- cycle;
    \filldraw[fill=white, rounded corners=2mm] (0.7,2.1) -- (1.8,3.3) -- (2.1,3.05) -- (0.9,1.7) -- cycle;
    \filldraw[fill=white] (2.05,3.3) circle (0.25);
    \begin{scope}[shift={(0,3.05)}, scale=0.75]
      \filldraw[fill=white] (-1.06,0) .. controls (-1.1,1.0) and (-0.6,1.25) .. (0,1.25)
        .. controls (0.6,1.25) and (1.1,1.0) .. (1.06,0) -- cycle;
      \orejasFW \caraFW
      \filldraw[fill=white] (-0.8,-0.62) .. controls (-0.85,-1.5) and (-0.3,-1.95) .. (0,-2.0)
        .. controls (0.3,-1.95) and (0.85,-1.5) .. (0.8,-0.62) .. controls (0.4,-0.85) and (-0.4,-0.85) .. (-0.8,-0.62) -- cycle;
      \begin{scope}[shift={(0,0.95)}] \coronaFW{0.8} \end{scope}
    \end{scope}
  \end{scope}
  % los caramelos, por el aire
  \foreach \x/\y/\a in {3.0/9.2/20, 4.4/8.3/-25, 5.4/9.6/35, 2.2/10.1/-10}
    {\begin{scope}[shift={(\x,\y)}, rotate=\a, scale=0.5]
      \filldraw[fill=white] (-0.6,0) -- (-1.05,0.4) -- (-1.0,-0.4) -- cycle;
      \filldraw[fill=white] (0.6,0) -- (1.05,0.4) -- (1.0,-0.4) -- cycle;
      \filldraw[fill=white] (0,0) ellipse (0.65 and 0.5);
    \end{scope}}
  \draw (-3.6,0) -- (6.0,0);
\end{tikzpicture}
