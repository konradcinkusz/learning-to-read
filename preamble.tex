% =========================================================================
% Tamaño de la frase de lectura, según el trimestre.
% Trimestre 1: una frase, muy grande. Trimestre 4: cuatro frases, algo
% más pequeñas para que quepan en el mismo bloque de la página.
% =========================================================================
\newcommand{\diafuente}[1]{%
  \ifcase#1\relax
  \or \fontsize{28}{33}\selectfont
  \or \fontsize{22}{27}\selectfont
  \or \fontsize{18}{23}\selectfont
  \or \fontsize{15}{20}\selectfont
  \fi
}

% =========================================================================
% La página de un día.
%
%   \begin{diapagina}{numero}{semana}{trimestre}
%     \begin{cajaLectura}
%     \centering\diafuente{trimestre}
%     Texto de lectura (una o varias frases).
%     \end{cajaLectura}
%     \vspace{5mm}
%     \actividadX{...}
%   \end{diapagina}
%
% \begin{diapagina} solo pone la cabecera (Día/Semana/Trimestre, Fecha,
% Notas) y \label{dia:N}; \end{diapagina} pone el pie y \newpage. Todo lo
% de en medio (la caja de lectura y la actividad) se escribe explícito,
% para que cada día pueda tener una actividad completamente distinta sin
% que la macro tenga que saber de antemano qué forma tiene.
% =========================================================================
\newenvironment{diapagina}[3]{%
  \label{dia:#1}%
  \def\diapagina@actual{#1}%
  \noindent{\LARGE\bfseries\lblDia~#1}\quad
  {\large(\lblSemana~#2 \textperiodcentered\ \lblTrimestre~#3)}\par
  \vspace{1mm}
  \textbf{\lblFecha:} \rule{32mm}{0.4pt}\hspace{10mm}%
  \textbf{\lblNotas:} \rule{55mm}{0.4pt}\par
  \vspace{4mm}
}{%
  \vfill
  \begin{center}\footnotesize\color{colorGris}\lblDia~\diapagina@actual~/ 260\end{center}
  \newpage
}
